```tsx
import React, { useEffect, useState } from 'react';

type LucideIconLike = React.ComponentType<{
    className?: string;
    size?: number | string;
    strokeWidth?: number;
    'aria-hidden'?: boolean | string;
}>;

// Module-level cache: shared across all instances of the component.
let cachedIcon: LucideIconLike | null = null;
let importAttempted = false;
let importPromise: Promise<LucideIconLike | null> | null = null;

function loadIcon(): Promise<LucideIconLike | null> {
    if (importAttempted) return Promise.resolve(cachedIcon);
    if (importPromise) return importPromise;
    importPromise = import('lucide-react')
        .then((m) => {
            cachedIcon = (m.Check ?? null) as LucideIconLike | null;
            importAttempted = true;
            return cachedIcon;
        })
        .catch(() => {
            importAttempted = true;
            cachedIcon = null;
            return null;
        });
    return importPromise;
}

function IconOrGlyph(...) {
    const [Icon, setIcon] = useState<LucideIconLike | null>(cachedIcon);
    useEffect(() => {
        if (importAttempted) return;
        let mounted = true;
        loadIcon().then((C) => { if (mounted) setIcon(C); });
        return () => { mounted = false; };
    }, []);
    if (Icon) return <Icon className={...} aria-hidden="true" />;
    return <span aria-hidden="true" data-testid="...-fallback-glyph">{GLYPH}</span>;
}
```

    ## Optional dependencies

    `@holmdigital/components` declares `lucide-react` as an OPTIONAL peer dependency. The four components that use Lucide icons — `Checkbox`, `HelpText`, `Select`, `Toast` — gracefully fall back to Unicode text glyphs when `lucide-react` is not installed.

    | `Toast` (info / success / warning / error) | `Info` / `CheckCircle` / `AlertTriangle` / `AlertCircle` | `ℹ` / `✓` / `⚠` / `⛔` |

    To enable Lucide icons in your application, install `lucide-react` alongside this package:

    ```bash
