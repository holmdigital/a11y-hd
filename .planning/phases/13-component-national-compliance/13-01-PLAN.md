// Replaces ENFORCEMENT_BODIES[country] lookup
export function getEnforcementBody(country: Country, sector?: 'public' | 'private'): string;
// sector omitted or 'public' -> WAD enforcement body
// sector = 'private' -> EAA enforcement body

// Returns national law or null for non-EU countries
export function getNationalLawByFramework(
    framework: 'WAD' | 'EAA',
    country: Country
): NationalLaw | null;

interface NationalLaw {
    law: string;       // e.g. "DOS-lagen"
    fullName: string;  // e.g. "Lag (2018:1937) om tillgänglighet till digital offentlig service"
}
